from pyscript import Element, when # Quiz questions questions = [ { "question": "What is the capital of Mada?", "options": ["Madagascar", "Antananarivo", "Mumbai", "Nairobi"], "answer": 1 }, { "question": "What is the largest city in Mada?", "options": ["Antananarivo", "Toamasina", "Mahajanga", "Fianarantsoa"], "answer": 0 }, { "question": "What is the official language of Mada?", "options": ["French", "English", "Malagasy", "Spanish"], "answer": 2 } ] current_question = 0 score = 0 def display_question(): global current_question q = questions[current_question] output = Element("quiz-output") output.clear() output.write(f"
{q['question']}
") for i, option in enumerate(q["options"]): output.write(f"
{option}
") for i in range(len(q["options"])): btn = Element(f"option-{i}") btn.add_event_listener("click", lambda event, idx=i: check_answer(idx)) def check_answer(selected_index): global current_question, score correct = questions[current_question]["answer"] == selected_index if correct: score += 1 current_question += 1 if current_question < len(questions): display_question() else: Element("quiz-output").write(f"
Quiz Complete! Your score: {score}/{len(questions)}
") @when("click", "#start-quiz") def start_quiz(event): Element("start-quiz").element.style.display = "none" display_question()
